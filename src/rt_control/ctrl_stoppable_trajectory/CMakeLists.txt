cmake_minimum_required(VERSION 3.16)
project(ctrl_stoppable_trajectory)
add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wshadow)
find_package(ament_cmake REQUIRED)
set(dependencies joint_trajectory_controller controller_interface hardware_interface pluginlib
  rclcpp rclcpp_action control_msgs trajectory_msgs lifecycle_msgs rt_control_interfaces robot_interfaces_qos)
foreach(dep IN LISTS dependencies)
  find_package(${dep} REQUIRED)
endforeach()
add_library(${PROJECT_NAME} SHARED src/controller.cpp)
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
target_compile_options(${PROJECT_NAME} PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
target_include_directories(${PROJECT_NAME} PUBLIC
  $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:include>)
ament_target_dependencies(${PROJECT_NAME} ${dependencies})
pluginlib_export_plugin_description_file(controller_interface plugins.xml)
install(TARGETS ${PROJECT_NAME} EXPORT export_${PROJECT_NAME} LIBRARY DESTINATION lib)
install(DIRECTORY include/ DESTINATION include)
install(FILES plugins.xml README.md DESTINATION share/${PROJECT_NAME})
ament_export_targets(export_${PROJECT_NAME} HAS_LIBRARY_TARGET)
ament_export_dependencies(${dependencies})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(controller_manager REQUIRED)
  ament_add_gtest(test_stop_controller test/test_stop_controller.cpp)
  target_link_libraries(test_stop_controller ${PROJECT_NAME})
  ament_target_dependencies(test_stop_controller controller_manager)
endif()
ament_package()

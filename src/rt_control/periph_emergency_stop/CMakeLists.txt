cmake_minimum_required(VERSION 3.8)
project(periph_emergency_stop)
add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Wshadow)
find_package(ament_cmake REQUIRED)
find_package(infra_modbus REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rt_control_interfaces REQUIRED)
add_executable(emergency_stop_node src/node.cpp)
target_compile_features(emergency_stop_node PRIVATE cxx_std_17)
ament_target_dependencies(emergency_stop_node infra_modbus rclcpp rt_control_interfaces)
install(TARGETS emergency_stop_node DESTINATION lib/${PROJECT_NAME})
install(FILES README.md DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_pytest_test(test_emergency_stop_node test/test_node.py TIMEOUT 90)
endif()
ament_package()
